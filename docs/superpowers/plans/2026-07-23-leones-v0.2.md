Hacer `PREY` public en `LionHuntPreyGoal` (`public static final Set<EntityType<?>> PREY = ...`). En `LionEntity` (imports: `com.gianni.entity.goal.LionHuntPreyGoal`, `net.minecraft.core.particles.ParticleTypes`, `net.minecraft.world.entity.Mob`, `net.minecraft.world.phys.AABB`):

```java
    private int roarCooldown;          // ticks; no persiste
    private boolean roaredTonight;

    public boolean canRoar() {
        return this.getVariant() == LionVariant.MALE && !this.isBaby();
    }

    public void roar() {
        if (this.level().isClientSide()) return;
        this.playSound(ModSounds.LION_ROAR.get(), 2.0f, 1.0f);
        if (this.level() instanceof ServerLevel server) {
            server.sendParticles(ParticleTypes.CLOUD,
                this.getX(), this.getEyeY(), this.getZ(), 8, 0.5, 0.3, 0.5, 0.02);
            for (Mob prey : server.getEntitiesOfClass(Mob.class,
                    this.getBoundingBox().inflate(16.0),
                    e -> LionHuntPreyGoal.PREY.contains(e.getType()))) {
                prey.setLastHurtByMob(this);   // dispara PanicGoal vanilla sin daño
            }
        }
        this.roarCooldown = 1200;
    }

    @Override
    public void aiStep() {
        super.aiStep();
        if (!this.level().isClientSide() && this.canRoar()) {
            if (this.roarCooldown > 0) this.roarCooldown--;
            boolean night = !this.isDaytime();
            if (!night) {
                this.roaredTonight = false;
            } else if (!this.roaredTonight) {
                this.roaredTonight = true;
                this.roar();
            }
            if (this.roarCooldown == 0 && this.getTarget() != null && this.getTarget().isAlive()) {
                this.roar();
            }
        }
    }
```

```java
package com.gianni.entity.goal;

import com.gianni.entity.LionEntity;
import net.minecraft.world.entity.ai.goal.Goal;

import java.util.EnumSet;
import java.util.List;

public class LionCubPlayGoal extends Goal {
    private final LionEntity cub;
    private final double speed;
    private LionEntity playmate;
    private int playTicks;

    public LionCubPlayGoal(LionEntity cub, double speed) {
        this.cub = cub;
        this.speed = speed;
        this.setFlags(EnumSet.of(Goal.Flag.MOVE, Goal.Flag.JUMP));
    }

    @Override
    public boolean canUse() {
        if (!cub.isBaby() || cub.getTarget() != null || cub.isInWater()) return false;
        if (cub.getRandom().nextInt(reducedTickDelay(60)) != 0) return false;
        List<LionEntity> cubs = cub.level().getEntitiesOfClass(
            LionEntity.class, cub.getBoundingBox().inflate(8.0),
            other -> other != cub && other.isBaby() && other.isAlive());
        if (cubs.isEmpty()) return false;
        this.playmate = cubs.get(cub.getRandom().nextInt(cubs.size()));
        return true;
    }

    @Override
    public boolean canContinueToUse() {
        return playTicks > 0 && playmate.isAlive() && cub.getTarget() == null;
    }

    @Override
    public void start() {
        this.playTicks = reducedTickDelay(120);
    }

    @Override
    public void tick() {
        playTicks--;
        cub.getNavigation().moveTo(playmate, speed);
        if (cub.distanceToSqr(playmate) < 4.0 && cub.onGround() && cub.getRandom().nextInt(5) == 0) {
            cub.getJumpControl().jump();
        }
    }

    @Override
    public void stop() {
        this.playmate = null;
        cub.getNavigation().stop();
    }
}
```

```java
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.CarpetBlock;
import net.minecraft.world.level.block.SoundType;
import net.minecraft.world.level.block.state.BlockBehaviour;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.BuildCreativeModeTabContentsEvent;
import net.neoforged.neoforge.registries.DeferredBlock;
import net.neoforged.neoforge.registries.DeferredRegister;

@EventBusSubscriber(modid = GianniMod.MODID)
public final class ModBlocks {
    public static final DeferredRegister.Blocks BLOCKS = DeferredRegister.createBlocks(GianniMod.MODID);

    public static final DeferredBlock<Block> LION_PELT_CARPET = registerCarpet("lion_pelt_carpet");
    public static final DeferredBlock<Block> WHITE_LION_PELT_CARPET = registerCarpet("white_lion_pelt_carpet");

    private static DeferredBlock<Block> registerCarpet(String name) {
        DeferredBlock<Block> block = BLOCKS.registerBlock(name, CarpetBlock::new,
            BlockBehaviour.Properties.of().strength(0.1f).sound(SoundType.WOOL));
        ModItems.ITEMS.registerSimpleBlockItem(block);
        return block;
    }

    private ModBlocks() {}

    @SubscribeEvent
    static void onBuildTabs(BuildCreativeModeTabContentsEvent event) {
        if (event.getTabKey() == CreativeModeTabs.FUNCTIONAL_BLOCKS) {
            event.accept(LION_PELT_CARPET);
            event.accept(WHITE_LION_PELT_CARPET);
        }
    }
}
```

```java
package com.gianni.worldgen;

import com.gianni.ModEntities;
import com.gianni.entity.LionEntity;
import net.minecraft.core.BlockPos;
import net.minecraft.util.RandomSource;
import net.minecraft.world.entity.EntitySpawnReason;
import net.minecraft.world.level.WorldGenLevel;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.levelgen.feature.Feature;
import net.minecraft.world.level.levelgen.feature.FeaturePlaceContext;
import net.minecraft.world.level.levelgen.feature.configurations.NoneFeatureConfiguration;

public class LionDenFeature extends Feature<NoneFeatureConfiguration> {

    public LionDenFeature() {
        super(NoneFeatureConfiguration.CODEC);
    }

    @Override
    public boolean place(FeaturePlaceContext<NoneFeatureConfiguration> ctx) {
        WorldGenLevel level = ctx.level();
        RandomSource random = ctx.random();
        BlockPos origin = ctx.origin();

        // planchón 12x12 orgánico, 2 capas
        for (int layer = 0; layer < 2; layer++) {
            int radius = 6 - layer * 2;
            for (int dx = -radius; dx <= radius; dx++) {
                for (int dz = -radius; dz <= radius; dz++) {
                    double dist = Math.sqrt(dx * dx + dz * dz);
                    if (dist > radius - random.nextDouble() * 1.5) continue;
                    BlockPos p = origin.offset(dx, layer, dz);
                    var block = random.nextInt(4) == 0 ? Blocks.SANDSTONE : Blocks.SMOOTH_SANDSTONE;
                    level.setBlock(p, block.defaultBlockState(), 3);
                }
            }
        }
        // huesos incrustados
        int bones = 3 + random.nextInt(3);
        for (int i = 0; i < bones; i++) {
            BlockPos p = origin.offset(random.nextInt(9) - 4, 0, random.nextInt(9) - 4);
            level.setBlock(p, Blocks.BONE_BLOCK.defaultBlockState(), 3);
        }
        // arbustos secos alrededor
        for (int i = 0; i < 5; i++) {
            BlockPos p = origin.offset(random.nextInt(15) - 7, 0, random.nextInt(15) - 7);
            if (level.isEmptyBlock(p.above(1)) && level.getBlockState(p).isSolid()) {
                level.setBlock(p.above(1), Blocks.DEAD_BUSH.defaultBlockState(), 3);
            }
        }
        // manada 2-4
        if (level.getLevel() != null) {
            int count = 2 + random.nextInt(3);
            for (int i = 0; i < count; i++) {
                LionEntity lion = ModEntities.LION.get().create(level.getLevel(), EntitySpawnReason.CHUNK_GENERATION);
                if (lion == null) continue;
                lion.snapTo(origin.getX() + random.nextInt(7) - 3 + 0.5,
                    origin.getY() + 2, origin.getZ() + random.nextInt(7) - 3 + 0.5,
                    random.nextFloat() * 360f, 0f);
                lion.finalizeSpawn(level.getLevel(), level.getCurrentDifficultyAt(origin),
                    EntitySpawnReason.CHUNK_GENERATION, null);
                level.addFreshEntity(lion);
            }
        }
        return true;
    }
}
```

// ADAPT: `snapTo` (renombre de `moveTo` en 26.x — probar ambos); `level.getLevel()` devuelve `ServerLevel` desde `WorldGenLevel` (si no existe, usar `level.getLevel()` de `WorldGenRegion` o castear con `instanceof ServerLevelAccessor` y `getLevel()`); `finalizeSpawn` con `getCurrentDifficultyAt` — si no está en WorldGenLevel, usar `level.getLevel().getCurrentDifficultyAt(...)`. `EntitySpawnReason.CHUNK_GENERATION` — si no existe, `NATURAL`.
