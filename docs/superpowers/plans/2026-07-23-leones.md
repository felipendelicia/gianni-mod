# Leones (Gianni Mod) — Plan de Implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Mod NeoForge para Minecraft 26.2 que agrega leones: hostiles/cazadores de noche, descansan de día, domesticables, montables y criables con variantes (macho/hembra/blanco).

**Architecture:** Un solo mob (`LionEntity extends TamableAnimal`) con goals de IA custom para caza nocturna y descanso diurno; montura vía `EquipmentSlot.SADDLE` + overrides de riding de vanilla; cliente con el pipeline nuevo de render states (`extractRenderState`/`submit`); assets (texturas PNG) generados por script Python.

**Tech Stack:** Java 25 · Gradle 9.2.1 (wrapper) · ModDevGradle 2.0.142 · NeoForge 26.2.0.28-beta · Minecraft 26.2 · Python 3 + Pillow (solo para generar texturas).

```java
import net.minecraft.world.entity.EntityType;
```

```java
import net.minecraft.world.entity.animal.Animal;
import net.minecraft.world.item.CreativeModeTabs;

import net.minecraft.world.level.levelgen.Heightmap;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.BuildCreativeModeTabContentsEvent;
import net.neoforged.neoforge.event.entity.EntityAttributeCreationEvent;
import net.neoforged.neoforge.event.entity.RegisterSpawnPlacementsEvent;
import net.neoforged.neoforge.registries.DeferredHolder;
import net.neoforged.neoforge.registries.DeferredItem;
import net.neoforged.neoforge.registries.DeferredRegister;
import net.minecraft.world.entity.SpawnPlacementTypes;

@EventBusSubscriber(modid = GianniMod.MODID)
public final class ModEntities {
    public static final DeferredRegister.Entities ENTITY_TYPES = DeferredRegister.createEntities(GianniMod.MODID);
    public static final DeferredRegister.Items ITEMS = DeferredRegister.createItems(GianniMod.MODID);

    public static final DeferredHolder<EntityType<?>, EntityType<LionEntity>> LION =
        ENTITY_TYPES.registerEntityType("lion", LionEntity::new, MobCategory.CREATURE,
            b -> b.sized(1.3f, 1.1f).eyeHeight(0.95f).clientTrackingRange(10));

    public static final DeferredItem<SpawnEggItem> LION_SPAWN_EGG =
        ITEMS.registerItem("lion_spawn_egg", props -> new SpawnEggItem(props.spawnEgg(LION.get())));

    private ModEntities() {}

    @SubscribeEvent
    static void onAttributes(EntityAttributeCreationEvent event) {
        event.put(LION.get(), LionEntity.createAttributes().build());
    }

    @SubscribeEvent
    static void onSpawnPlacements(RegisterSpawnPlacementsEvent event) {
        event.register(LION.get(), SpawnPlacementTypes.ON_GROUND,
            Heightmap.Types.MOTION_BLOCKING_NO_LEAVES, Animal::checkAnimalSpawnRules,
            RegisterSpawnPlacementsEvent.Operation.REPLACE);
    }

    @SubscribeEvent
    static void onBuildTabs(BuildCreativeModeTabContentsEvent event) {
        if (event.getTabKey() == CreativeModeTabs.SPAWN_EGGS) {
            event.accept(LION_SPAWN_EGG);
        }
    }
}
```

```java
package com.gianni.entity.goal;

import com.gianni.entity.LionEntity;
import net.minecraft.world.entity.ai.goal.Goal;

import java.util.EnumSet;

public class LionRestGoal extends Goal {
    private final LionEntity lion;

    public LionRestGoal(LionEntity lion) {
        this.lion = lion;
        this.setFlags(EnumSet.of(Goal.Flag.MOVE, Goal.Flag.JUMP));
    }

    @Override
    public boolean canUse() {
        return !lion.isTame() && lion.isDaytime() && lion.getTarget() == null && !lion.isInLove();
    }

    @Override
    public boolean canContinueToUse() {
        return canUse();
    }

    @Override
    public void start() {
        lion.getNavigation().stop();
        lion.setResting(true);
    }

    @Override
    public void stop() {
        lion.setResting(false);
    }
}
```

```python
#!/usr/bin/env python3
"""Genera las texturas del león. Las regiones corresponden a los texOffs de LionModel.
Layout de un box WxHxD en texOffs(u,v): franja superior (tapas) en y=[v, v+D),
franja inferior (caras) en y=[v+D, v+D+H); caras: [D][W][D][W] de ancho."""
from PIL import Image, ImageDraw
from pathlib import Path

OUT = Path(__file__).resolve().parent.parent / "src/main/resources/assets/gianni/textures"

PALETTES = {
    "lion_male":   {"base": (198, 154, 98), "dark": (150, 110, 66), "mane": (107, 74, 38)},
    "lion_female": {"base": (198, 154, 98), "dark": (150, 110, 66), "mane": (150, 110, 66)},
    "lion_white":  {"base": (232, 228, 218), "dark": (190, 186, 176), "mane": (200, 196, 186)},
}
BLACK = (30, 24, 18)
PINK = (120, 70, 60)


def box_region(d, u, v, w, h, dep, base, dark):
    # franja de tapas + franja de caras, con leve variación para que no sea plano
    d.rectangle([u + dep, v, u + dep + 2 * w - 1, v + dep - 1], fill=dark)          # tapas
    d.rectangle([u, v + dep, u + 2 * (dep + w) - 1, v + dep + h - 1], fill=base)    # caras
    d.rectangle([u, v + dep, u + dep - 1, v + dep + h - 1], fill=dark)              # cara lateral sombreada


def paint(name, pal):
    im = Image.new("RGBA", (64, 64), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    box_region(d, 0, 0, 8, 8, 6, pal["base"], pal["dark"])       # cabeza
    box_region(d, 28, 0, 4, 3, 3, pal["base"], pal["dark"])      # hocico
    box_region(d, 28, 6, 2, 2, 1, pal["dark"], pal["dark"])      # oreja izq
    box_region(d, 34, 6, 2, 2, 1, pal["dark"], pal["dark"])      # oreja der
    box_region(d, 40, 10, 11, 11, 3, pal["mane"], pal["mane"])   # melena
    box_region(d, 0, 20, 10, 10, 16, pal["base"], pal["dark"])   # cuerpo
    box_region(d, 0, 46, 3, 10, 3, pal["base"], pal["dark"])     # pata A
    box_region(d, 12, 46, 3, 10, 3, pal["base"], pal["dark"])    # pata B
    box_region(d, 56, 0, 1, 9, 1, pal["dark"], pal["dark"])      # cola
    box_region(d, 36, 46, 11, 3, 8, (90, 60, 40), (60, 40, 25))  # montura
    # cara frontal de la cabeza: box (0,0) 8x8x6 -> cara frontal en (6,6)..(13,13)
    d.point([(8, 9), (11, 9)], fill=BLACK)                        # ojos
    d.rectangle([9, 11, 10, 12], fill=PINK)                       # nariz
    (OUT / "entity/lion").mkdir(parents=True, exist_ok=True)
    im.save(OUT / f"entity/lion/{name}.png")


def egg():
    im = Image.new("RGBA", (16, 16), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    d.ellipse([3, 2, 12, 13], fill=(198, 154, 98))
    d.ellipse([5, 5, 7, 7], fill=(107, 74, 38))
    d.ellipse([8, 9, 10, 11], fill=(107, 74, 38))
    (OUT / "item").mkdir(parents=True, exist_ok=True)
    im.save(OUT / "item/lion_spawn_egg.png")


for name, pal in PALETTES.items():
    paint(name, pal)
egg()
print("ok")
```

```java
package com.gianni;

import com.gianni.entity.LionEntity;
import com.gianni.entity.LionVariant;
import net.minecraft.core.BlockPos;
import net.minecraft.core.registries.Registries;
import net.minecraft.gametest.framework.GameTestHelper;
import net.neoforged.neoforge.registries.DeferredRegister;

import java.util.function.Consumer;
import java.util.function.Supplier;

public final class ModGameTests {
    public static final DeferredRegister<Consumer<GameTestHelper>> TEST_FUNCTIONS =
        DeferredRegister.create(Registries.TEST_FUNCTION, GianniMod.MODID);
    // ADAPT: nombre del registry key de funciones de test según docs 26.2

    public static final Supplier<Consumer<GameTestHelper>> LION_SIT =
        TEST_FUNCTIONS.register("lion_sit", () -> helper -> {
            LionEntity lion = helper.spawnMob(ModEntities.LION.get(), new BlockPos(1, 2, 1));
            lion.setTame(true, false);
            lion.setOrderedToSit(true);
            helper.succeedWhen(() -> {
                if (!lion.isInSittingPose()) helper.fail("el león no entró en pose de sentado");
            });
        });

    public static final Supplier<Consumer<GameTestHelper>> LION_BREED_VARIANT =
        TEST_FUNCTIONS.register("lion_breed_variant", () -> helper -> {
            LionEntity a = helper.spawnMob(ModEntities.LION.get(), new BlockPos(1, 2, 1));
            LionEntity b = helper.spawnMob(ModEntities.LION.get(), new BlockPos(2, 2, 1));
            a.setVariant(LionVariant.MALE);
            b.setVariant(LionVariant.FEMALE);
            var cub = (LionEntity) a.getBreedOffspring(helper.getLevel(), b);
            if (cub == null) helper.fail("getBreedOffspring devolvió null");
            LionVariant v = cub.getVariant();
            if (v != LionVariant.MALE && v != LionVariant.FEMALE && v != LionVariant.WHITE) {
                helper.fail("variante inválida: " + v);
            }
            cub.discard();
            helper.succeed();
        });

    private ModGameTests() {}
}
```
