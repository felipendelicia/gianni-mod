# Leones v0.3.0 — Plan de Implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Yuyos cultivables que drogan leones (catnip + dopaje) y overhaul gráfico del león (modelo ~25 piezas, texturas painter, ojos emisivos, mandíbula al rugir).

**Architecture:** Los yuyos son dos `CropBlock` estilo beetroot (4 etapas, item=semilla) cuyo uso sobre `LionEntity` activa un estado synched (`TRIPPING`+`LionCatnipGoal`) o MobEffects vanilla. El overhaul gráfico reescribe `LionModel` (código, sin libs) con atlas 128x128 generado por `gen_textures.py` reescrito como painter, con **gate de preview aprobado por Felipe antes de integrar**.

**Tech Stack:** NeoForge 26.2.0.28-beta / MC 26.2, Java 25, Gradle (`./gradlew`), Python 3 + Pillow para texturas, GameTests data-driven (`test_instance` JSON + `Registries.TEST_FUNCTION`).

**Spec:** `docs/superpowers/specs/2026-07-24-leones-v0.3-design.md`

package com.gianni.block;

import com.mojang.serialization.MapCodec;
import net.minecraft.world.level.block.CropBlock;
import net.minecraft.world.level.block.state.BlockState;
import net.minecraft.world.level.block.state.StateDefinition;
import net.minecraft.world.level.block.state.properties.BlockStateProperties;
import net.minecraft.world.level.block.state.properties.IntegerProperty;
import net.minecraft.world.level.ItemLike;

import java.util.function.Supplier;

public class YuyoCropBlock extends CropBlock {
    public static final int MAX_AGE = 3;
    public static final IntegerProperty AGE = BlockStateProperties.AGE_3;
    // ADAPT: MapCodec — si CropBlock exige codec() abstracto, usar simpleCodec con un supplier fijo por bloque
    public static final MapCodec<YuyoCropBlock> CODEC = simpleCodec(p -> new YuyoCropBlock(() -> net.minecraft.world.item.Items.WHEAT_SEEDS, p));

    private final Supplier<? extends ItemLike> seed;

    public YuyoCropBlock(Supplier<? extends ItemLike> seed, Properties props) {
        super(props);
        this.seed = seed;
    }

    @Override public MapCodec<? extends CropBlock> codec() { return CODEC; }
    @Override public IntegerProperty getAgeProperty() { return AGE; }
    @Override public int getMaxAge() { return MAX_AGE; }
    @Override protected ItemLike getBaseSeedId() { return seed.get(); }

    @Override
    protected void createBlockStateDefinition(StateDefinition.Builder<net.minecraft.world.level.block.Block, BlockState> builder) {
        builder.add(AGE);
    }
}

package com.gianni.entity.goal;

import com.gianni.entity.LionEntity;
import net.minecraft.world.entity.ai.goal.Goal;

import java.util.EnumSet;

public class LionCatnipGoal extends Goal {
    private final LionEntity lion;
    private int tick;

    public LionCatnipGoal(LionEntity lion) {
        this.lion = lion;
        this.setFlags(EnumSet.of(Goal.Flag.MOVE, Goal.Flag.JUMP, Goal.Flag.LOOK));
    }

    @Override public boolean canUse() { return lion.isTripping(); }
    @Override public boolean requiresUpdateEveryTick() { return true; }
    @Override public void start() { this.tick = 0; }

    @Override
    public void tick() {
        tick++;
        int phase = (tick / 60) % 3;             // 3 s por fase: rodar, círculos, saltitos
        if (phase == 0) {
            lion.getNavigation().stop();
            lion.setRolling(true);
        } else {
            lion.setRolling(false);
            if (phase == 1) {
                if (lion.getNavigation().isDone()) {
                    double ang = (tick % 120) / 120.0 * Math.PI * 2;
                    lion.getNavigation().moveTo(lion.getX() + Math.cos(ang) * 4.0, lion.getY(),
                        lion.getZ() + Math.sin(ang) * 4.0, 1.3);
                }
            } else if (lion.onGround() && lion.getRandom().nextInt(10) == 0) {
                lion.getJumpControl().jump();
            }
        }
    }

    @Override
    public void stop() {
        lion.setRolling(false);
        lion.getNavigation().stop();
    }
}

    this.head.yRot = s.yRot * Mth.DEG_TO_RAD;
    this.head.xRot = s.xRot * Mth.DEG_TO_RAD;
    boolean mane = s.male && !s.isBaby;
    this.maneInner.visible = mane;
    this.maneMid.visible = mane;
    this.maneOuter.visible = mane;
    this.saddle.visible = s.saddled;
    this.collar.visible = s.collared && !s.isBaby;
    this.jaw.xRot = s.roaring ? 0.55f : 0.0f;

    float pos = s.walkAnimationPos;
    float speed = s.walkAnimationSpeed;
    this.rightFrontUpper.xRot = Mth.cos(pos * 0.6662f) * 1.2f * speed;
    this.leftHindUpper.xRot = this.rightFrontUpper.xRot;
    this.leftFrontUpper.xRot = Mth.cos(pos * 0.6662f + Mth.PI) * 1.2f * speed;
    this.rightHindUpper.xRot = this.leftFrontUpper.xRot;
    this.tail1.xRot = 0.6f + Mth.cos(pos * 0.3331f) * 0.25f * speed;
    this.tail1.yRot = Mth.sin(s.ageInTicks * 0.08f) * 0.25f + Mth.sin(pos * 0.6662f) * 0.3f * speed;
    this.tail2.xRot = 0.2f + Mth.cos(pos * 0.3331f + 0.5f) * 0.15f * speed;
    this.earR.zRot = -0.3f + Mth.sin(s.ageInTicks * 0.05f) * 0.05f;
    this.earL.zRot = 0.3f - Mth.sin(s.ageInTicks * 0.05f + 1.7f) * 0.05f;

    if (s.rolling) {                      // panza arriba, patas al aire
        this.bodyFront.zRot = Mth.PI;
        this.bodyBack.zRot = Mth.PI;
        this.belly.zRot = Mth.PI;
        this.bodyFront.y += 6.0f;
        this.bodyBack.y += 6.0f;
        this.belly.y -= 4.0f;
        this.head.zRot = Mth.PI * 0.9f;
        this.head.y += 8.0f;
        float w = Mth.sin(s.ageInTicks * 0.5f) * 0.5f;
        this.rightFrontUpper.xRot = -2.5f + w;
        this.leftFrontUpper.xRot = -2.5f - w;
        this.rightHindUpper.xRot = -2.5f - w;
        this.leftHindUpper.xRot = -2.5f + w;
        this.rightFrontUpper.y += 6.0f;
        this.leftFrontUpper.y += 6.0f;
        this.rightHindUpper.y += 6.0f;
        this.leftHindUpper.y += 6.0f;
        return;                           // rolling pisa stalking/sitting
    }

    if (s.stalking) {
        this.bodyFront.y += 3.0f;
        this.bodyBack.y += 3.0f;
        this.belly.y += 3.0f;
        this.head.y += 4.5f;
        this.head.xRot += 0.2f;
        this.rightFrontUpper.y += 3.0f;
        this.leftFrontUpper.y += 3.0f;
        this.rightHindUpper.y += 3.0f;
        this.leftHindUpper.y += 3.0f;
    }

    if (s.sitting || s.resting) {
        this.bodyFront.y += 5.0f;
        this.bodyBack.y += 5.0f;
        this.belly.y += 5.0f;
        this.head.y += 5.0f;
        this.rightFrontUpper.xRot = -Mth.HALF_PI;
        this.leftFrontUpper.xRot = -Mth.HALF_PI;
        this.rightHindUpper.xRot = -Mth.HALF_PI;
        this.leftHindUpper.xRot = -Mth.HALF_PI;
        this.rightFrontUpper.y += 5.0f;
        this.leftFrontUpper.y += 5.0f;
        this.rightHindUpper.y += 5.0f;
        this.leftHindUpper.y += 5.0f;
        if (s.resting) {
            this.head.y += 2.0f;
            this.head.xRot += 0.3f;
        }
    }
}

ATLAS = 128

def ramp(base, n=5, spread=0.42):
    """n tonos del más oscuro al más claro alrededor de base."""
    out = []
    for i in range(n):
        f = 1.0 - spread + (2 * spread) * i / (n - 1)
        out.append(tuple(min(255, int(c * f)) for c in base))
    return out

RAMPS = {
    "lion_male":   {"fur": ramp((198, 154, 98)), "mane": ramp((107, 74, 38)), "belly": ramp((222, 196, 156))},
    "lion_female": {"fur": ramp((204, 162, 108)), "mane": ramp((150, 110, 66)), "belly": ramp((226, 202, 164))},
    "lion_white":  {"fur": ramp((232, 228, 218)), "mane": ramp((200, 196, 186)), "belly": ramp((244, 242, 236))},
}
BLACK = (30, 24, 18)
PINK = (198, 120, 110)
EYE_AMBER = (216, 168, 60)

def fur_box(d, u, v, w, h, dep, r, rng, stroke=True):
    """Pinta un box con sombreado por cara + trazos de pelaje.
    r: rampa de 5 tonos. Tapa superior clara, inferior oscura, frente medio,
    laterales medio-oscuro, con strokes verticales +-1 tono."""
    top, bottom = r[3], r[0]
    d.rectangle([u + dep, v, u + dep + w - 1, v + dep - 1], fill=top)                    # tapa sup
    d.rectangle([u + dep + w, v, u + dep + 2 * w - 1, v + dep - 1], fill=bottom)         # tapa inf
    faces = [(u, dep, r[1]), (u + dep, w, r[2]), (u + dep + w, dep, r[1]), (u + dep + w + dep, w, r[2])]
    for fx, fw, tone in faces:
        d.rectangle([fx, v + dep, fx + fw - 1, v + dep + h - 1], fill=tone)
        if stroke:
            for sx in range(fx, fx + fw):
                if rng.random() < 0.45:
                    sy = v + dep + rng.randrange(max(1, h))
                    ln = rng.randrange(1, 3)
                    idx = 1 if rng.random() < 0.5 else 3
                    d.line([(sx, sy), (sx, min(v + dep + h - 1, sy + ln))], fill=r[idx])
    # oclusión abajo de cada cara
    d.rectangle([u, v + dep + h - 1, u + 2 * (dep + w) - 1, v + dep + h - 1], fill=r[0])

package com.gianni.client;

import com.gianni.GianniMod;
import com.mojang.blaze3d.vertex.PoseStack;
import com.mojang.blaze3d.vertex.VertexConsumer;
import net.minecraft.client.renderer.MultiBufferSource;
import net.minecraft.client.renderer.RenderType;
import net.minecraft.client.renderer.entity.RenderLayerParent;
import net.minecraft.client.renderer.entity.layers.RenderLayer;
import net.minecraft.client.renderer.texture.OverlayTexture;

public class LionEyesLayer extends RenderLayer<LionRenderState, LionModel> {
    private static final RenderType EYES = RenderType.eyes(GianniMod.id("textures/entity/lion/lion_eyes.png"));

    public LionEyesLayer(RenderLayerParent<LionRenderState, LionModel> parent) {
        super(parent);
    }

    @Override
    public void render(PoseStack pose, MultiBufferSource buffers, int packedLight,
            LionRenderState s, float yRot, float xRot) {
        if (!s.nightEyes) return;
        VertexConsumer vc = buffers.getBuffer(EYES);
        this.getParentModel().renderToBuffer(pose, vc, 0xF000F0, OverlayTexture.NO_OVERLAY);
    }
}
